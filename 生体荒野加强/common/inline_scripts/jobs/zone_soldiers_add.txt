### Regular Empires
# Planets
triggered_district_planet_modifier = {
	potential = {
		exists = owner
		owner = {
			is_gestalt = no
		}
		planet_is_habitat_equivalent = no
		planet_is_ecu_equivalent = no
		planet_is_ring_world_equivalent = no
        planet_is_wilderness_update_planet = no
		NOT = { has_modifier = slave_colony }
		NOT = { has_knights_zone = yes }
	}
	job_soldier_add = $AMOUNT$
}

# Habitats
triggered_district_planet_modifier = {
	potential = {
		exists = owner
		owner = {
			is_gestalt = no
		}
		planet_is_habitat_equivalent = yes
		NOT = { has_modifier = slave_colony }
		NOT = { has_knights_zone = yes }
	}
	job_soldier_add = $AMOUNT$
	mult = owner.value:hab_void_dweller_jobs
}
