ap_hydrocentric = {
    potential = {
        has_aquatics = yes
        NOT = { has_ascension_perk = ap_hydrocentric }
        OR = {
            has_trait = trait_pc_ocean_preference
            has_trait = trait_wet_planet_preference
            has_trait = trait_robot_aquatic
            has_trait = trait_auto_hab_preference
        }
        is_nomadic = no
    }
    possible = {
        has_technology = tech_terrestrial_sculpting
        OR = {
            has_aquatic_trait = yes
            has_trait = trait_wet_planet_preference
            has_trait = trait_auto_hab_preference
        }
    }

    modifier = {
        terraforming_ocean_cost_mult = -0.25
        add_attunement_the_composer_of_strands = @ascension_perk_attunement1
    }

    on_enabled = {
        custom_tooltip = ap_hydrocentric_tooltip
        if = {
            limit = {
                host_has_dlc = "Apocalypse"
            }
            if = {
                limit = { has_ascension_perk = ap_colossus }
                custom_tooltip = ap_hydrocentric_deluge_tooltip
            }
            else = {
                custom_tooltip = ap_hydrocentric_deluge_pre_unlock_tooltip
            }
        }
        hidden_effect = {
            country_event = { id = aquatics.1 }
        }
    }

    tradition_swap = {
        name = ap_hydrocentric_wilderness
        inherit_effects = no
        inherit_icon = yes
        trigger = {
            is_wilderness_empire = yes
        }

        modifier = {
            terraforming_ocean_cost_mult = -0.25
            add_attunement_the_composer_of_strands = @ascension_perk_attunement1
        }

        on_enabled = {
            custom_tooltip = ap_hydrocentric_wilderness_tt
            if = {
                limit = {
                    host_has_dlc = "Apocalypse"
                }
                if = {
                    limit = { has_ascension_perk = ap_colossus }
                    custom_tooltip = ap_hydrocentric_deluge_tooltip
                }
                else = {
                    custom_tooltip = ap_hydrocentric_deluge_pre_unlock_tooltip
                }
            }
            hidden_effect = {
                country_event = { id = aquatics.1 }
            }
        }
    }

    ai_weight = {
        factor = 80 #it really is quite good for Aquatic species
        modifier = {
            factor = 0.5
            is_xenophobe = no
            is_hive_empire = no
        }
        modifier = {
            factor = 5
            has_origin = origin_ocean_paradise
        }
        modifier = {
            factor = 5
            has_origin = origin_ocean_machines
        }
    }
}

ap_world_shaper = {
    possible = {
        has_technology = tech_ecological_adaptation
        custom_tooltip = {
            fail_text = "requires_ascension_perks_1"
            num_ascension_perks > 0
        }
    }

    potential = {
        is_nomadic = no
        NOT = {
            has_ascension_perk = ap_world_shaper
        }
        OR = {
            is_hive_empire = no
            is_wilderness_empire = yes
        }
        OR = {
            is_machine_empire = no
            has_valid_civic = civic_machine_servitor
            has_valid_civic = civic_machine_assimilator
        }
    }

    modifier = {
        terraforming_cost_mult = -0.25

    }

    on_enabled = {
        custom_tooltip = "allow_gaia"
    }

    tradition_swap = {
        name = ap_world_shaper_wilderness
        inherit_effects = no
        inherit_icon = yes
        trigger = {
            is_wilderness_empire = yes
        }

        modifier = {
            terraforming_cost_mult = -0.25

                has_ethic = ethic_gestalt_consciousness
            }
        }
        modifier = {
            factor = 2
            OR = {
                has_ethic = ethic_pacifist
                has_ethic = ethic_xenophobe
            }
        }
        modifier = {
            factor = 3
            OR = {
                has_ethic = ethic_fanatic_pacifist
                has_ethic = ethic_fanatic_xenophobe
                is_wilderness_empire = yes
            }
        }
        modifier = { #only way to get more nice planets
            factor = 100
            has_origin = origin_life_seeded
        }
    }
}
